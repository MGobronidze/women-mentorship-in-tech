Fetch სრულად, JSON, DevTools და Weather App

წინა გაკვეთილში ვისწავლეთ, როგორ ვმართავთ ასინქრონულობას. ახლა ეს ცოდნა რეალურ სცენარში გამოვიყენოთ — მონაცემები წამოვიღოთ სერვერიდან, დავამუშაოთ, გამოვსახოთ, და ბოლოს ავაწყოთ სრული ამინდის აპლიკაცია რეალური, საჯარო API-ით.

1. AJAX — კონცეფცია

AJAX (Asynchronous JavaScript and XML) არის ტექნიკის საერთო სახელი, რომელიც JavaScript-ს საშუალებას აძლევს, სერვერს მოსთხოვოს მონაცემები გვერდის ხელახლა ჩატვირთვის გარეშე. სახელწოდებაში „XML" ისტორიულია — დღეს თითქმის ყოველთვის JSON-ს ვიყენებთ, XML-ის ნაცვლად.

2. XMLHttpRequest — ისტორიული საფუძველი

XMLHttpRequest (XHR) იყო პირველი API ბრაუზერის AJAX მოთხოვნებისთვის. ის callback-ებზეა აწყობილი (onload, onerror), ამიტომ რთულ სცენარებში ადვილად ვარდება წინა გაკვეთილში ნანახ „Callback Hell"-ში — სწორედ ამიტომ შეიქმნა Promise-ზე დაფუძნებული fetch. დღეს XHR იშვიათად იწერება ხელით, მაგრამ ბევრ ძველ კოდში ჯერ კიდევ გვხვდება: 

const xhr = new XMLHttpRequest();
xhr.open("GET", "https://jsonplaceholder.typicode.com/users");
xhr.onload = function() {
  if (xhr.status === 200) {
    console.log(JSON.parse(xhr.responseText));
  }
};
xhr.onerror = function() {
  console.error("ქსელის შეცდომა");
};
xhr.send();

3. Fetch API — თანამედროვე სტანდარტი

fetch() არის Promise-ზე დაფუძნებული, ბევრად მარტივი API HTTP მოთხოვნების გასაგზავნად. დღეს ეს არის სტანდარტული არჩევანი — და წინა გაკვეთილზე ნასწავლი async/await მასთან იდეალურად მუშაობს. 

// GET მოთხოვნა — async/await-ით
async function loadUsers() {
  try {
    const response = await fetch("https://jsonplaceholder.typicode.com/users");

    if (!response.ok) {
      throw new Error(`HTTP შეცდომა: ${response.status}`);
    }

    const users = await response.json(); // პასუხის ტექსტის JSON-ად გარდაქმნა
    console.log(users);
  } catch (error) {
    console.error("მონაცემების წამოღება ვერ მოხერხდა:", error.message);
  }
}

⚠️ ყველაზე ხშირი დამწყები შეცდომა — fetch HTTP შეცდომაზე „არ ვარდება"

fetch()-ის Promise reject-დება მხოლოდ მაშინ, როცა მოთხოვნა საერთოდ ვერ შესრულდა (არ არის ინტერნეტი, DNS ვერ მოიძებნა, CORS-ით დაიბლოკა). თუ სერვერმა უპასუხა — თუნდაც 404-ით ან 500-ით — Promise წარმატებით სრულდება, და catch ბლოკი არ ამუშავდება: 

const response = await fetch("https://jsonplaceholder.typicode.com/users/9999");

console.log(response.ok);     // false — სტატუსი 200-299 დიაპაზონში არ არის
console.log(response.status); // 404

// catch-ბლოკი აქ არ ამუშავდებოდა! ამიტომ response.ok ხელით უნდა შევამოწმოთ:
if (!response.ok) {
  throw new Error(`HTTP ${response.status}`);
}

Response ობიექტი — რას შეიცავს

  • response.ok — true, თუ სტატუსი 200-დან 299-მდეა.
  • response.status — რიცხვითი კოდი (200, 404, 500...).
  • response.headers.get("content-type") — პასუხის header-ები.
  • response.json() / .text() / .blob() — სხეულის წაკითხვა სხვადასხვა ფორმატში (ყველა Promise-ს აბრუნებს).

მნიშვნელოვანი დეტალი: პასუხის სხეული შეიძლება წაიკითხო მხოლოდ ერთხელ. თუ ჯერ response.text() გამოიძახე, მერე response.json() შეცდომას გამოიწვევს. თუ ორივე ფორმატი გჭირდება — response.clone()-ით გააკეთე ასლი.

URL-ის აწყობა — URLSearchParams

მოთხოვნის პარამეტრების (query string) სტრიქონების ხელით შეერთება ("?name=" + name) სახიფათოა — სივრცეები, & სიმბოლო და ქართული ასოები URL-ში სწორად უნდა დაიშიფროს (percent-encoding). URLSearchParams ამას ავტომატურად აკეთებს: 

const params = new URLSearchParams({ name: "თბილისი", count: 1 });

console.log(params.toString());
// "name=%E1%83%97%E1%83%91%E1%83%98%E1%83%9A%E1%83%98%E1%83%A1%E1%83%98&count=1"

const url = `https://geocoding-api.open-meteo.com/v1/search?${params}`; // ${params} თავად გადაიქცევა სტრიქონად

// სივრცეები და სპეც. სიმბოლოებიც უსაფრთხოდ მუშავდება:
new URLSearchParams({ q: "new york & co" }).toString(); // "q=new+york+%26+co"

POST მოთხოვნა — მონაცემების გაგზავნა

async function createPost(postData) {
  const response = await fetch("https://jsonplaceholder.typicode.com/posts", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(postData), // JS ობიექტი → JSON სტრიქონი
  });

  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return response.json(); // სერვერი დაგვიბრუნებს შექმნილ ობიექტს, id-ითურთ
}

// ფორმიდან მონაცემების ამოღება — FormData + Object.fromEntries (მე-12 გაკვეთილი)
form.addEventListener("submit", async (event) => {
  event.preventDefault();
  const data = Object.fromEntries(new FormData(form)); // { title: "...", body: "..." }
  const created = await createPost(data);
  console.log("შეიქმნა:", created);
});

მოთხოვნის გაუქმება და timeout — AbortController

fetch თავისით არასდროს წყვეტს ლოდინს — თუ სერვერი „გაჩერდა", მოთხოვნა შეიძლება დიდხანს „გაიჭედოს". სამი ხელსაწყო გვეხმარება (პირველს მე-16 გაკვეთილზე უკვე შევხვდით event listener-ების კონტექსტში): 

// 1. ხელით გაუქმება — AbortController
const controller = new AbortController();
fetch(url, { signal: controller.signal });
controller.abort(); // fetch დაუყოვნებლივ reject-დება; error.name === "AbortError"

// 2. ავტომატური timeout — AbortSignal.timeout(მილიწამები)
const response = await fetch(url, { signal: AbortSignal.timeout(5000) });
// 5 წამში პასუხი თუ არ მოვიდა, error.name === "TimeoutError"

// 3. ორივე ერთად — AbortSignal.any([...]): პირველივე გაუქმების მიზეზი იმარჯვებს
const signal = AbortSignal.any([controller.signal, AbortSignal.timeout(5000)]);
await fetch(url, { signal });

თავსებადობა (2026): AbortSignal.timeout() „Baseline Newly available" სტატუსშია 2024 წლის აპრილიდან, და web-features-ის პროგნოზით 2026 წლის 18 ოქტომბერს „Widely available"-ს გახდება. AbortSignal.any() მუშაობს Chrome 116+, Firefox 124+ და Safari 17.4+-ში — ძველ ბრაუზერებზე მის გარეშე გამოიყენე ცალკე timeout ან controller.

შეცდომების სახეობები — რას „ვხედავთ" catch-ში

რა ხდება როგორ ამოვიცნობთ რას ვუთხრათ მომხმარებელს
ქსელი/DNS/CORS ჩავარდა TypeError, error.status არ არსებობს „შეამოწმე ინტერნეტი"
სერვერმა შეცდომით უპასუხა response.ok === false → ჩვენი error.status 404 → „ვერ მოიძებნა", 5xx → „სერვერის პრობლემაა"
timeout error.name === "TimeoutError" „სერვერი დიდხანს არ პასუხობს"
ჩვენ თვითონ გავაუქმეთ error.name === "AbortError" არაფერი — ეს შეცდომა არ არის
პასუხი არ არის ვალიდური JSON SyntaxError (response.json()-იდან) „მონაცემები არასწორი ფორმატითაა"

Race Condition — „მოძველებული პასუხის" პრობლემა

წარმოიდგინე: მომხმარებელმა მოძებნა „Batumi", მაშინვე — „Tbilisi". თუ პირველი მოთხოვნა ნელია და მეორეზე გვიან მოვიდა, ეკრანზე „Tbilisi"-ს ნაცვლად „Batumi"-ს შედეგი გამოჩნდება — ე.წ. stale response. გამოსავალი: ახალი მოთხოვნის დაწყებისას წინა controller.abort()-ით გავაუქმოთ (ზუსტად ამას ქვემოთ, Weather App-ში გავაკეთებთ).

CORS და API გასაღებები — უსაფრთხოება

  • CORS (Cross-Origin Resource Sharing): ბრაუზერი ბლოკავს სხვა დომენის პასუხის წაკითხვას, თუ სერვერი პასუხში არ აბრუნებს Access-Control-Allow-Origin header-ს. Console-ში ამას „blocked by CORS policy" წერია. ეს frontend-იდან გამოსასწორებელი არ არის — სერვერმა უნდა დაუშვას (ან შენი საკუთარი backend-ის გავლით გააგზავნო მოთხოვნა). Open-Meteo და JSONPlaceholder CORS-ს უშვებენ.
  • საიდუმლო გასაღებები (API key, token) არასდროს ჩაწერო frontend კოდში — ყველაფერი, რაც ბრაუზერშია, ხილულია DevTools-ში ნებისმიერისთვის. ფასიანი/პირადი API-ებისთვის გასაღები backend-ზე უნდა ინახებოდეს. სწორედ ამიტომ არის Open-Meteo მისაღები სასწავლო პროექტისთვის: მას გასაღები საერთოდ არ სჭირდება.

პარალელური მოთხოვნები (წინა გაკვეთილის გამეორება)

ერთმანეთისგან დამოუკიდებელი მოთხოვნები პარალელურად გავუშვათ Promise.all()-ით (ან Promise.allSettled()-ით, თუ ერთის ჩავარდნა დანარჩენებს არ უნდა აფუჭებდეს). ხოლო ერთმანეთზე დამოკიდებული მოთხოვნები (მაგ. ჯერ ქალაქის კოორდინატები, შემდეგ — ამინდი) თანმიმდევრულ await-ს მოითხოვს.

4. JSON — ობიექტი და მეთოდები

JSON (JavaScript Object Notation) არის ტექსტური ფორმატი მონაცემების გადასაცემად — ის ჰგავს JS ობიექტს, მაგრამ არის მკაცრად ტექსტური (string). 

const user = { name: "ანა", age: 22, active: true };

// JS ობიექტი → JSON სტრიქონი (გასაგზავნად)
const jsonString = JSON.stringify(user);
console.log(jsonString); // '{"name":"ანა","age":22,"active":true}'

// JSON სტრიქონი → JS ობიექტი (მიღებული პასუხის დასამუშავებლად)
const parsedUser = JSON.parse(jsonString);
console.log(parsedUser.name); // "ანა"

// ლამაზი, დაშორებებიანი ფორმატი (დებაგირებისთვის): მესამე არგუმენტი — დაშორების ზომა
console.log(JSON.stringify(user, null, 2));

response.json(), რომელსაც Fetch-თან ერთად ვიყენებთ, სინამდვილეში JSON.parse()-ის „მოხერხებული" ვერსიაა, უშუალოდ HTTP პასუხისთვის.

JSON ≠ JS ობიექტი — რას კარგავს stringify

  • undefined, ფუნქციები და Symbol ობიექტიდან უბრალოდ იკარგება.
  • Date გადაიქცევა ISO სტრიქონად — parse-ის შემდეგ ის უკვე სტრიქონია, არა Date.
  • NaN და Infinity გადაიქცევა null-ად.
  • JSON-ის წესები უფრო მკაცრია: key-ები ორმაგ ბრჭყალებში, ბოლო ელემენტის შემდეგ მძიმე დაუშვებელია, კომენტარები არ არსებობს.
  • ამიტომ JSON.parse(JSON.stringify(obj)) ობიექტის კოპირების არასანდო გზაა — სანაცვლოდ გამოიყენე structuredClone() (მე-12 გაკვეთილი).
// JSON.parse არავალიდურ ტექსტზე შეცდომას ისვრის — ყოველთვის try/catch-ში:
try {
  const data = JSON.parse(text);
} catch (error) {
  console.error("არავალიდური JSON:", error.message); // SyntaxError
}

5. DevTools — Network ტაბი

Network ტაბი (გავეცანით 1-ლ გაკვეთილში ზოგადად) fetch მოთხოვნების დებაგირებისთვის უმთავრესი ინსტრუმენტია. ყველაზე სასარგებლო შესაძლებლობები:

  • Fetch/XHR ფილტრი — დამალავს სურათებს, ფონტებს, CSS-ს და დაგტოვებს მხოლოდ API მოთხოვნებს.
  • მოთხოვნაზე დაწკაპუნება → Headers (რა გაგზავნე), Payload (body), Response და Preview (გაფორმებული ხედი), Timing (სად დაიხარჯა დრო: ლოდინი სერვერზე vs ჩამოტვირთვა).
  • Throttling (ქსელის შენელება: Slow 4G, 3G, Offline — სახელები Chrome-ის ვერსიით ოდნავ განსხვავდება) — ერთადერთი სწორი გზა, რომ შენი loading და error მდგომარეობები რეალურად გამოსცადო.
  • Block request URL (მარჯვენა კლავიში მოთხოვნაზე) — კონკრეტული მოთხოვნის „ჩავარდნის" სიმულაცია, სერვერის გათიშვის გარეშე.
  • Copy → Copy as fetch — ნებისმიერი ქსელური მოთხოვნის ზუსტი ასლი JS კოდად, Console-ში გასამეორებლად.
  • Disable cache და Preserve log — ხელს უშლის ძველი პასუხების მოტყუებას და გვერდის გადატვირთვისას ჩანაწერებს არ შლის.

6. საკონტროლო პროექტი — Weather App

ახლა ავაწყოთ სრული ამინდის აპლიკაცია, რომელიც იყენებს 1-დან 19-ე გაკვეთილამდე ნასწავლ ყველა ძირითად კონცეფციას. მონაცემებისთვის ვიყენებთ Open-Meteo-ს — უფასო, საჯარო ამინდის API-ს, რომელსაც გასაღები არ სჭირდება (არაკომერციული გამოყენებისთვის, დღეში 10 000 მოთხოვნამდე) და CORS-ს უშვებს, ამიტომ პირდაპირ ბრაუზერიდან მუშაობს.

რას გააკეთებს აპლიკაცია

  • ქალაქის სახელით ძებნა → ამჟამინდელი ამინდი (ტემპერატურა, შეგრძნება, ტენიანობა, ქარი) + 5-დღიანი პროგნოზი;
  • ოთხი მკაფიო მდგომარეობა: idle, loading, success, error;
  • წინა ძებნის გაუქმება ახალი ძებნისას (stale response-ის გარეშე) და 8-წამიანი timeout;
  • „ფავორიტი ქალაქების" პანელი — სამი ქალაქი პარალელურად იტვირთება, ერთის ჩავარდნა დანარჩენებს არ აფუჭებს.

ორი API მოთხოვნა — ერთმანეთზე დამოკიდებული

  1. Geocoding: geocoding-api.open-meteo.com/v1/search?name=... — ქალაქის სახელი → კოორდინატები. თუ ვერაფერი მოიძებნა, პასუხში results ველი საერთოდ არ არის!
  2. Forecast: api.open-meteo.com/v1/forecast?latitude=...&longitude=... — კოორდინატები → ამინდი. პასუხში weather_code არის WMO სტანდარტული კოდი (0 = მოწმენდილი, 61 = წვიმა, 95 = ჭექა-ქუხილი...), რომელსაც ჩვენ თვითონ გადავათარგმნით ემოჯიდ და ქართულ ტექსტად.

მეორე მოთხოვნა პირველის შედეგზეა დამოკიდებული — ამიტომ აქ თანმიმდევრული await სწორი არჩევანია. ფავორიტი ქალაქების კოორდინატები კი წინასწარ ვიცით, ამიტომ მათი ამინდი დამოუკიდებელი მოთხოვნებია — პარალელურად წავიღოთ.

js/api.js — ქსელური ფენა

ერთი დამხმარე ფუნქცია (fetchJSON) ერთ ადგილას ამოწმებს response.ok-ს, ხოლო searchCity/getForecast მხოლოდ URL-ს აგებენ. error.status-ს ვინახავთ, რომ UI-მა მოგვიანებით შეცდომა კატეგორიებად დაყოს. 

/* api.js — ქსელთან კომუნიკაცია (Fetch). DOM-ს არ ეხება. */

const GEO_URL = "https://geocoding-api.open-meteo.com/v1/search";
const FORECAST_URL = "https://api.open-meteo.com/v1/forecast";

// fetch() მხოლოდ ქსელის შეცდომაზე "ვარდება" — HTTP 404/500-ზე არა! ამიტომ ვამოწმებთ response.ok-ს.
async function fetchJSON(url, signal) {
  const response = await fetch(url, { signal });

  if (!response.ok) {
    const error = new Error(`HTTP ${response.status}`);
    error.status = response.status; // კატეგორიზაციისთვის, UI-ში
    throw error;
  }

  return response.json();
}

// ნაბიჯი 1: ქალაქის სახელი → კოორდინატები
export async function searchCity(name, signal) {
  // URLSearchParams თავად აკეთებს encoding-ს — ქართული ასოებიც უსაფრთხოდ გადადის URL-ში
  const params = new URLSearchParams({ name, count: "1", language: "en", format: "json" });
  const data = await fetchJSON(`${GEO_URL}?${params}`, signal);

  // თუ ვერაფერი მოიძებნა, API-ს პასუხში "results" ველი საერთოდ არ არის!
  if (!data.results || data.results.length === 0) {
    return null;
  }

  const { name: cityName, latitude, longitude, country } = data.results[0];
  return { name: cityName, latitude, longitude, country };
}

// ნაბიჯი 2: კოორდინატები → ამინდის პროგნოზი
export function getForecast({ latitude, longitude }, signal) {
  const params = new URLSearchParams({
    latitude,
    longitude,
    current: "temperature_2m,apparent_temperature,relative_humidity_2m,wind_speed_10m,weather_code",
    daily: "weather_code,temperature_2m_max,temperature_2m_min",
    timezone: "auto",
    forecast_days: "5",
  });
  return fetchJSON(`${FORECAST_URL}?${params}`, signal);
}

js/weather.js — წმინდა ლოგიკა

ეს ფაილი არც DOM-ს ეხება, არც ქსელს — მხოლოდ ფუნქციები, რომლებიც მონაცემს იღებენ და მონაცემს აბრუნებენ (ისინი ადვილად ტესტირებადია). API-ს „ნედლი" პასუხი (temperature_2m, daily.time[]...) გადადის ჩვენთვის მოსახერხებელ სტრუქტურაში — ასე ქსელური ფორმატის ცვლილება მხოლოდ ერთ ფაილს შეეხება. 

/* weather.js — წმინდა ლოგიკა (pure functions): კოდები, ნორმალიზაცია, შეცდომების აღწერა */

// WMO weather code → ემოჯი და ქართული აღწერა
const WEATHER_CODES = [
  { codes: [0], emoji: "☀️", label: "მოწმენდილი ცა" },
  { codes: [1], emoji: "🌤️", label: "ძირითადად მოწმენდილი" },
  { codes: [2], emoji: "⛅", label: "ნაწილობრივ მოღრუბლული" },
  { codes: [3], emoji: "☁️", label: "მოღრუბლული" },
  { codes: [45, 48], emoji: "🌫️", label: "ნისლი" },
  { codes: [51, 53, 55, 56, 57], emoji: "🌦️", label: "წვრილი წვიმა" },
  { codes: [61, 63, 65, 66, 67], emoji: "🌧️", label: "წვიმა" },
  { codes: [71, 73, 75, 77], emoji: "🌨️", label: "თოვლი" },
  { codes: [80, 81, 82], emoji: "🌦️", label: "კოკისპირული წვიმა" },
  { codes: [85, 86], emoji: "🌨️", label: "თოვლის ჩავლა" },
  { codes: [95, 96, 99], emoji: "⛈️", label: "ჭექა-ქუხილი" },
];

export function getWeatherInfo(code) {
  const match = WEATHER_CODES.find((item) => item.codes.includes(code));
  return match ? { emoji: match.emoji, label: match.label } : { emoji: "❓", label: "უცნობი" };
}

// API-ს "ნედლი" პასუხი → ჩვენთვის მოსახერხებელი სტრუქტურა
export function normalizeForecast(raw) {
  const { current, daily } = raw;

  return {
    current: {
      temp: Math.round(current.temperature_2m),
      feelsLike: Math.round(current.apparent_temperature),
      humidity: current.relative_humidity_2m,
      wind: Math.round(current.wind_speed_10m),
      ...getWeatherInfo(current.weather_code),
    },
    days: daily.time.map((date, i) => ({
      date,
      max: Math.round(daily.temperature_2m_max[i]),
      min: Math.round(daily.temperature_2m_min[i]),
      ...getWeatherInfo(daily.weather_code[i]),
    })),
  };
}

// "2026-10-14" → "ოთხ" (ქართული ლოკალი, Intl API)
export function formatWeekday(isoDate) {
  // "T00:00:00" — რომ თარიღი ადგილობრივ დროდ ჩაითვალოს, არა UTC-დ
  return new Date(`${isoDate}T00:00:00`).toLocaleDateString("ka-GE", { weekday: "short" });
}

// შეცდომა → მომხმარებლისთვის გასაგები ტექსტი
export function describeError(error) {
  if (error.name === "TimeoutError") return "სერვერმა დიდხანს არ უპასუხა. სცადე ხელახლა.";
  if (error.status === 429) return "ძალიან ბევრი მოთხოვნა. ცოტა ხანში სცადე.";
  if (error.status >= 500) return "სერვერის შეცდომა. მოგვიანებით სცადე.";
  if (error.status) return `მოთხოვნა ვერ შესრულდა (HTTP ${error.status}).`;
  return "ქსელთან კავშირი ვერ დამყარდა. შეამოწმე ინტერნეტი.";
}

js/ui.js — მხოლოდ DOM

render(state) — ერთადერთი ფუნქცია, რომელიც state-ს ეკრანად გარდაქმნის. ყურადღება მიაქციე: API-დან მოსულ ტექსტს (ქალაქის სახელი) ვწერთ textContent-ით, არა innerHTML-ით (მე-15 გაკვეთილის XSS გაფრთხილება) — თუ API-მ ან მომხმარებელმა <img onerror=...> დაგვიბრუნა, ის უბრალო ტექსტად დარჩება. 

/* ui.js — მხოლოდ DOM. API-ს "ნედლ" მონაცემებს არ იცნობს — ღებულობს უკვე ნორმალიზებულს. */
import { formatWeekday } from "./weather.js";

// დამხმარე: ელემენტის შექმნა. textContent — უსაფრთხოა (API-დან მოსული ტექსტი HTML-ად არ პარსირდება!)
function el(tag, className, text) {
  const node = document.createElement(tag);
  if (className) node.className = className;
  if (text !== undefined) node.textContent = text;
  return node;
}

function createCurrentCard(city, current) {
  const card = el("section", "weather-current");
  card.append(
    el("h2", "weather-city", `${city.name}${city.country ? ", " + city.country : ""}`),
    el("p", "weather-emoji", current.emoji),
    el("p", "weather-temp", `${current.temp}°C`),
    el("p", "weather-label", current.label),
    el("p", "weather-meta", `შეგრძნება ${current.feelsLike}° · ტენიანობა ${current.humidity}% · ქარი ${current.wind} კმ/სთ`)
  );
  return card;
}

function createForecastList(days) {
  const list = el("ul", "forecast-grid");
  const fragment = document.createDocumentFragment(); // მე-15 გაკვეთილი

  for (const { date, emoji, max, min, label } of days) {
    const item = el("li", "forecast-day");
    item.title = label;
    item.append(
      el("span", "forecast-weekday", formatWeekday(date)),
      el("span", "forecast-emoji", emoji),
      el("span", "forecast-temps", `${max}° / ${min}°`)
    );
    fragment.appendChild(item);
  }

  list.appendChild(fragment);
  return list;
}

// ერთი ფუნქცია, რომელიც state-ს გარდაქმნის ეკრანად
export function render(state, container) {
  container.replaceChildren(); // ძველი შიგთავსის გასუფთავება
  container.dataset.state = state.status;

  switch (state.status) {
    case "idle":
      container.append(el("p", "weather-hint", "შეიყვანე ქალაქის სახელი და დააჭირე „ძებნას“."));
      break;
    case "loading":
      container.append(el("p", "weather-loading", "⏳ იტვირთება..."));
      break;
    case "error":
      container.append(el("p", "weather-error", `⚠️ ${state.message}`));
      break;
    case "success":
      container.append(createCurrentCard(state.city, state.forecast.current), createForecastList(state.forecast.days));
      break;
  }
}

// ფავორიტი ქალაქების პანელი (Promise.allSettled-ის შედეგებიდან)
export function renderFavorites(items, container) {
  container.replaceChildren();

  for (const { city, forecast } of items) {
    const row = el("li", "favorite-city");
    row.append(
      el("span", "favorite-name", city.name),
      forecast
        ? el("span", "favorite-temp", `${forecast.current.emoji} ${forecast.current.temp}°C`)
        : el("span", "favorite-temp favorite-failed", "— მიუწვდომელია")
    );
    container.appendChild(row);
  }
}

js/app.js — ყველაფრის შეკრება

აქ ერთად მუშაობს: state (ერთი ობიექტი, ერთი setState), event (submit + preventDefault), თანმიმდევრული await (ძებნა → პროგნოზი), AbortController + AbortSignal.any/timeout (გაუქმება და timeout), და Promise.allSettled (ფავორიტები). 

/* app.js — "შემკრები": state, events, async ლოგიკა */
import { searchCity, getForecast } from "./api.js";
import { normalizeForecast, describeError } from "./weather.js";
import { render, renderFavorites } from "./ui.js";

const FAVORITES = [
  { name: "თბილისი", latitude: 41.72, longitude: 44.79 },
  { name: "ბათუმი", latitude: 41.65, longitude: 41.64 },
  { name: "ქუთაისი", latitude: 42.27, longitude: 42.7 },
];

const form = document.querySelector("#search-form");
const input = document.querySelector("#city-input");
const output = document.querySelector("#weather-output");
const favoritesList = document.querySelector("#favorites-list");

let state = { status: "idle" };
let currentController = null; // მიმდინარე მოთხოვნის გასაუქმებლად

function setState(newState) {
  state = newState;
  render(state, output);
}

async function handleSearch(cityName) {
  // 1. წინა, ჯერ კიდევ მიმდინარე მოთხოვნა უქმდება (stale response-ის პრობლემა)
  currentController?.abort();
  currentController = new AbortController();

  // 2. ერთი signal: ხელით გაუქმება + 8 წამიანი timeout
  const signal = AbortSignal.any([currentController.signal, AbortSignal.timeout(8000)]);

  setState({ status: "loading" });

  try {
    const city = await searchCity(cityName, signal);
    if (!city) {
      setState({ status: "error", message: `„${cityName}“ ვერ მოიძებნა. სცადე სხვა დაწერილობა.` });
      return;
    }

    const raw = await getForecast(city, signal); // დამოკიდებულია city-ზე → თანმიმდევრული await
    setState({ status: "success", city, forecast: normalizeForecast(raw) });
  } catch (error) {
    if (error.name === "AbortError") return; // ახალმა ძებნამ გააუქმა — ეს შეცდომა არ არის
    setState({ status: "error", message: describeError(error) });
  }
}

form.addEventListener("submit", (event) => {
  event.preventDefault();
  const cityName = input.value.trim();
  if (cityName.length < 2) {
    setState({ status: "error", message: "შეიყვანე მინიმუმ 2 სიმბოლო." });
    return;
  }
  handleSearch(cityName);
});

// ფავორიტები — დამოუკიდებელი მოთხოვნები, ამიტომ პარალელურად; allSettled — ერთის ჩავარდნა დანარჩენს არ აფუჭებს
async function loadFavorites() {
  const results = await Promise.allSettled(
    FAVORITES.map((city) => getForecast(city, AbortSignal.timeout(8000)))
  );

  const items = results.map((result, i) => ({
    city: FAVORITES[i],
    forecast: result.status === "fulfilled" ? normalizeForecast(result.value) : null,
  }));

  renderFavorites(items, favoritesList);
}

render(state, output);
loadFavorites();
