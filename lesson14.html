ეს გაკვეთილი გვასწავლის თანამედროვე JavaScript-ის (და React-ის) ყველაზე ხშირად გამოყენებულ სინტაქსურ ხერხებს — და ბოლოს, ამ ცოდნით, საბოლოოდ დავშლით პორტფოლიოს პროექტს სუფთა, პროფესიონალურ მოდულებად.

1. Rest Parameter

...ოპერატორი ფუნქციის პაარამეტრებს ერთ მასივად აგროვებს: 

function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}

sum(1, 2);        // 3
sum(1, 2, 3, 4);  // 10 — ნებისმიერი რაოდენობის არგუმენტი

// rest ყოველთვის ბოლოშია, სხვა პარამეტრებთან ერთად:
function introduce(first, second, ...others) {
  console.log(first, second, others);
}
introduce("ა", "ბ", "გ", "დ"); // "ა" "ბ" ["გ", "დ"]

Rest-ის კიდევ ერთი გამოყენება — destructuring-ში (მე-3 სექციაში დეტალურად ვნახავთ), სადაც ის „აგროვებს" დარჩენილ ელემენტებს/თვისებებს.

2. Spread Operator

იგივე ... სინტაქსი, შებრუნებული დანიშნულება — „შლის" მასივს ან ობიექტს ცალკეულ ელემენტებად: 

// მასივების გაერთიანება/კოპირება
const fruits = ["ვაშლი", "მსხალი"];
const moreFruits = [...fruits, "ბანანი"]; // ["ვაშლი", "მსხალი", "ბანანი"]

// ობიექტების გაერთიანება/კოპირება
const student = { name: "ანა", age: 22 };
const updated = { ...student, age: 23 }; // { name: "ანა", age: 23 } — ხელახლა ჩაწერილი "age" გადაფარავს ძველს

// ფუნქციის გამოძახებისას მასივის "დაშლა" ცალკეულ არგუმენტებად
const nums = [4, 9, 16];
Math.max(...nums); // 16

React-ში spread ხშირად გამოიყენება ობიექტის ან მასივის განახლებისთვის ისე, რომ ძველი მონაცემი პირდაპირ არ შევცვალოთ.

Spread + Set — მასივის დუბლიკატების მოცილება

Set არის ჩაშენებული სტრუქტურა, რომელიც მხოლოდ უნიკალურ მნიშვნელობებს ინახავს — spread-თან კომბინაციაში იგი ხშირად გამოიყენება დუბლიკატების სწრაფად მოსაცილებლად: 

const tags = ["JS", "CSS", "JS", "HTML", "CSS"];

const uniqueTags = [...new Set(tags)];
console.log(uniqueTags); // ["JS", "CSS", "HTML"] — დუბლიკატების გარეშე, თანმიმდევრობა შენარჩუნებულია

მნიშვნელოვანი შენიშვნა — Spread აკეთებს მხოლოდ Shallow კოპირებას

Spread ქმნის ახალ მასივს/ობიექტს, მაგრამ მხოლოდ ერთ დონეზე — თუ ობიექტის შიგნით არის ჩადგმული ობიექტი/მასივი, ის კვლავ იმავე მისამართს „უყურებს", ორიგინალთან ერთად: 

const project = { title: "საიტი", meta: { year: 2025 } };
const copy = { ...project };

copy.title = "ახალი სახელი";   // ✅ ორიგინალს არ ეხება — ეს ზედა დონეზეა
copy.meta.year = 2026;          // ⚠️ ორიგინალსაც ცვლის! meta ორივეში იგივე ობიექტია

console.log(project.meta.year); // 2026 — მოულოდნელად შეიცვალა!

ღრმად ჩადგმული ობიექტების ნამდვილი, დამოუკიდებელი ასლისთვის (deep copy) — გამოიყენე structuredClone() (მე-12 გაკვეთილიდან ნაცნობი), არა უბრალო spread.

3. მასივის და ობიექტის Destructuring

Destructuring საშუალებას გვაძლევს, მასივის ან ობიექტის მნიშვნელობები პირდაპირ ცალკეულ ცვლადებში შევინახოთ (ინდექსით ან key-ით მიმართვის გარეშე). 

// მასივის destructuring — პოზიციის მიხედვით
const coordinates = [42.31, 43.36];
const [lat, lng] = coordinates;
console.log(lat, lng); // 42.31 43.36

// ობიექტის destructuring — key-ის სახელის მიხედვით
const student = { name: "ანა", age: 22, city: "თბილისი" };
const { name, age } = student;
console.log(name, age); // "ანა" 22

// გადარქმევა და ნაგულისხმევი მნიშვნელობა
const { name: studentName, country = "საქართველო" } = student;
console.log(studentName, country); // "ანა" "საქართველო" (country არ არსებობდა, ნაგულისხმევი გამოიყენა)

Rest დესტრუქტურირებაში — „დანარჩენის" აღება

const { name: n, ...restInfo } = student;
console.log(n);        // "ანა"
console.log(restInfo);  // { age: 22, city: "თბილისი" } — ყველაფერი, name-ის გარდა

const [first, ...rest] = [1, 2, 3, 4];
console.log(first, rest); // 1 [2, 3, 4]

ჩადგმული (Nested) Destructuring

ობიექტის/მასივის სტრუქტურა შეიძლება „გამეორდეს" destructuring-შიც, ნებისმიერ სიღრმეზე:

const project = {
  title: "პორტფოლიოს საიტი",
  meta: { year: 2026, tags: ["HTML", "CSS"] },
};

const {
  title,
  meta: { year, tags: [firstTag] }, // ჩადგმული ობიექტიდან, ჩადგმული მასივის პირველი ელემენტი
} = project;

console.log(title, year, firstTag); // "პორტფოლიოს საიტი" 2026 "HTML"

ცვლადების გაცვლა (Swap) — Destructuring-ის კლასიკური ტრიკი

let a = 1;
let b = 2;

[a, b] = [b, a]; // დროებითი ცვლადის გარეშე!
console.log(a, b); // 2 1

Destructuring ფუნქციის პარამეტრებში

Destructuring განსაკუთრებით ხშირად გამოიყენება ფუნქციის პარამეტრებში — React-ის კომპონენტში props-ის მისაღებად ეს არის სტანდარტული პრაქტიკა: 

// ობიექტის destructuring პირდაპირ პარამეტრებში, default მნიშვნელობასთან ერთად
function printStudent({ name, age = "უცნობია" }) {
  console.log(`${name}, ${age} წლის`);
}
printStudent(student);

4. მოდულური JavaScript — import/export

დღეს პროექტები იშლება ცალკეულ ფაილებად (მოდულებად), რომლებიც ერთმანეთს export/import-ით უზიარებენ ცვლადებს, ფუნქციებს და კლასებს. 

// math.js — ექსპორტი
export function add(a, b) {
  return a + b;
}
export const PI = 3.14159;

export default function multiply(a, b) { // default export — ერთ ფაილში მხოლოდ ერთი
  return a * b;
}
// app.js — იმპორტი
import multiply, { add, PI } from "./math.js";

console.log(add(2, 3));      // 5
console.log(multiply(2, 3)); // 6
console.log(PI);              // 3.14159

named export (export function add()) შესაძლებელია ერთ ფაილში ბევრი გვქონდეს და იმპორტისას ფიგურული ფრჩხილებით ავირჩიოთ საჭირო ნაწილები; default export კი ერთადერთია ფაილში და იმპორტისას ნებისმიერი სახელით შეიძლება ჩამოვწეროთ. ეს არის ზუსტად ის სისტემა, რომელსაც React-ის კომპონენტების ორგანიზებისას გამოვიყენებთ.

<script type="module"> — HTML-ში მოდულის გამოყენება

<script type="module" src="app.js"></script>

type="module" აუცილებელია, რომ ბრაუზერმა import/export სინტაქსი ამოიცნოს.

Dynamic import() — მოდულის „მოთხოვნისამებრ" ჩატვირთვა

ჩვეულებრივი import ფაილის თავში იწერება და ყოველთვის, მაშინვე იტვირთება. Dynamic import() (ფუნქციის მსგავსი სინტაქსით) საშუალებას გვაძლევს, მოდული ჩავტვირთოთ მხოლოდ მაშინ, როცა ის რეალურად დაგვჭირდება — ეს არის საფუძველი React-ისა და Vite-ის „code splitting"-ის (მოგვიანებით ვნახავთ), რაც საწყისი გვერდის ჩატვირთვის სისწრაფეს მნიშვნელოვნად აუმჯობესებს დიდ აპლიკაციებში: 

async function loadChart() {
  const { renderChart } = await import("./chart.js"); // მხოლოდ საჭიროების შემთხვევაში ჩაიტვირთება
  renderChart();
}

document.querySelector("#show-chart-btn").addEventListener("click", loadChart);
// chart.js საერთოდ არ ჩაიტვირთება, სანამ მომხმარებელი ღილაკზე არ დააჭერს

5. საკონტროლო პროექტი — პორტფოლიოს დაშლა მოდულებად

წინა ორ შეხვედრაზე (მე-12, მე-13) ავაწყვეთ პორტფოლიოს გვერდი, გამოვყავით მონაცემები და ფუნქციები — მაგრამ ყველაფერი ჯერ კიდევ ერთ, დიდ data.js ფაილშია. ახლა, მოდულების ცოდნით, ეს პროექტი საბოლოოდ სამ, სუფთა, ერთმანეთისგან დამოუკიდებელ ფაილად დავშალოთ — ზუსტად ისე, როგორც რეალურ პროექტებში ხდება.

ახალი ფაილური სტრუქტურა

lesson12-portfolio/
├── index.html          <!-- <script type="module" src="app.js"> -->
├── style.css
├── data.js               // მხოლოდ მონაცემები — projects მასივი, export-ით
├── project-utils.js       // მხოლოდ ფუნქციები — import მონაცემებზე დამოკიდებული არ არის
└── app.js                 // "შემკრები" —  უშვებს ლოგიკას
// data.js — მხოლოდ მონაცემები
export const projects = [
  { title: "To-Do აპლიკაცია", year: 2025, tags: ["JS", "CSS"], featured: true },
  { title: "პორტფოლიოს საიტი", year: 2026, tags: ["HTML", "CSS"], featured: true },
  { title: "ამინდის ვიჯეტი", year: 2025, tags: ["JS", "API"], featured: false },
  { title: "ბლოგის დიზაინი", year: 2024, tags: ["CSS", "Figma"], featured: false },
];
// project-utils.js — ფუნქციები, destructuring/rest/spread-ის გამოყენებით

export function getFeaturedProjects(projectList) {
  return projectList.filter((p) => p.featured);
}

// Rest parameter — ნებისმიერი რაოდენობის tag-ის მიხედვით ფილტრი
export function getProjectsByAnyTag(projectList, ...searchTags) {
  return projectList.filter((p) =>
    p.tags.some((tag) => searchTags.includes(tag))
  );
}

// Destructuring პირდაპირ პარამეტრში
export const formatProjectSummary = ({ title, year }) => `${title} (${year})`;

// Spread + Set — ყველა უნიკალური tag, ყველა პროექტიდან ერთად
export function getAllUniqueTags(projectList) {
  const allTags = projectList.flatMap((p) => p.tags); // ყველა tag, ერთ "ბრტყელ" მასივში
  return [...new Set(allTags)];
}

// Spread — ახალი პროექტის დამატება, ორიგინალის შეცვლის გარეშე (immutable)
export function addProject(projectList, newProject) {
  return [...projectList, newProject];
}
// app.js — "შემკრები", ყველაფერს აერთიანებს
import { projects } from "./data.js";
import {
  getFeaturedProjects,
  getProjectsByAnyTag,
  formatProjectSummary,
  getAllUniqueTags,
  addProject,
} from "./project-utils.js";

console.log("გამორჩეული:", getFeaturedProjects(projects).map(formatProjectSummary));
console.log("JS/API ტეგიანი:", getProjectsByAnyTag(projects, "JS", "API").map(formatProjectSummary));
console.log("ყველა უნიკალური tag:", getAllUniqueTags(projects));

const updatedProjects = addProject(projects, {
  title: "ახალი პროექტი", year: 2026, tags: ["React"], featured: false,
});
console.log("ორიგინალის სიგრძე:", projects.length);         // 4 — უცვლელია!
console.log("განახლებულის სიგრძე:", updatedProjects.length); // 5

index.html-ში საჭიროა მხოლოდ ერთი ცვლილება — <script> ტეგს დაემატოს type="module" და src გადავიდეს app.js-ზე: 

<script type="module" src="app.js"></script>

რატომ არის ეს სტრუქტურა უკეთესი: თითოეულ ფაილს ერთი, მკაფიო პასუხისმგებლობა აქვს — data.js არაფერს „აკეთებს", მხოლოდ ინახავს; project-utils.js არაფერს „იცის" კონკრეტულად ჩვენს მონაცემებზე, უბრალოდ იღებს ნებისმიერ სიას პარამეტრად; app.js კი მხოლოდ „აკავშირებს" ყველაფერს. ეს არის ზუსტად ის არქიტექტურული აზროვნება, რომელსაც React-ის მოდულში (კომპონენტების, hooks-ისა და utils-ის გამიჯვნისას) გამოვიყენებთ.

  2. რატომ იცვლება ორიგინალი ობიექტიც, თუ spread-ით შექმნილი ასლის ჩადგმულ თვისებას ვცვლით?

  4. რას აკეთებს [a, b] = [b, a] — რატომ არ სჭირდება დროებითი ცვლადი?
  5. რა განსხვავებაა ჩვეულებრივ import-სა და dynamic import()-ს შორის — როდის გამოვიყენებდით მეორეს?
  6. რატომ არის data.js/project-utils.js/app.js-ად დაყოფა უკეთესი, ვიდრე ყველაფრის ერთ ფაილში ქონა?

პრაქტიკული დავალებები

  1. დაწერე ფუნქცია rest parameter-ით, რომელიც ნებისმიერი რაოდენობის სტრიქონს ერთმანეთს „აწებებს" space-ით.
  2. ორი მასივი გააერთიანე ერთში spread ოპერატორის გამოყენებით, Set-ის დახმარებით დუბლიკატების გარეშე.
  3. დაწერე კოდი, რომელიც აჩვენებს spread-ის „shallow copy" პრობლემას — ჩადგმული ობიექტის შეცვლით, რომელიც ორიგინალსაც ცვლის.
  4. შექმენი ობიექტი 4 თვისებით და destructuring-ის გამოყენებით ამოიღე 2 კონკრეტული ცვლადი, დანარჩენები კი — rest-ის მეშვეობით ერთ ობიექტში ({ a, b, ...rest }).
  5. დაწერე ჩადგმული ობიექტი (მინიმუმ 2 დონე) და ერთ destructuring გამოსახულებაში ამოიღე ცვლადი ორივე დონიდან.
  6. გამოსცადე ცვლადების გაცვლა ([a, b] = [b, a]) და Console-ში დაადასტურე, რომ იმუშავა.
  7. დაწერე ფუნქცია, რომელიც პარამეტრად იღებს ობიექტს პირდაპირ destructuring-ით (მაგ. { title, price }) და აბრუნებს ფორმატირებულ სტრიქონს.
  8. (საკონტროლო პროექტი) გახსენი მე-12/13 გაკვეთილის პორტფოლიოს პროექტი და დაშალე ის სამ ფაილად — data.js (named export projects-ით), project-utils.js (named export ფუნქციებით, დესტრუქტურირების/rest/spread-ის გამოყენებით) და app.js (ორივეს იმპორტავს). განაახლე index.html-ის <script> ტეგი type="module"-ით და Live Server-ით შეამოწმე, რომ Console-ის შედეგი იგივეა, რაც აქამდე.
