OOP, Prototype, Classes, Storage & Cookies

JavaScript მოდულის ბოლო გაკვეთილი — ობიექტზე ორიენტირებული პროგრამირების საფუძვლები და ბრაუზერში მონაცემების შენახვის გზები. ეს ცოდნა პირდაპირ მოგვამზადებს React-ისთვის.

1. Constructor Function

Constructor function არის „ქარხანა", საიდანაც ერთი და იმავე „ფორმის" ბევრი ობიექტის შექმნა შეიძლება. Convention-ის მიხედვით, სახელი იწერება დიდი ასოთი და გამოიძახება new საკვანძო სიტყვით: 

function Student(name, age) {
  this.name = name;
  this.age = age;
}

const student1 = new Student("ანა", 22);
const student2 = new Student("გიორგი", 24);

console.log(student1.name); // "ანა"

2. Prototype და მემკვიდრეობითობა

ყველა ობიექტს JavaScript-ში აქვს „ფარული" კავშირი მშობელ ობიექტთან — prototype-თან. მეთოდები, საერთო ყველა instance-სთვის, საუკეთესოა prototype-ზე დავდოთ (და არა constructor-ის შიგნით) — ეს ეკონომიურია მეხსიერების თვალსაზრისით: 

Student.prototype.introduce = function() {
  return `გამარჯობა, მე ვარ ${this.name}`;
};

console.log(student1.introduce()); // "გამარჯობა, მე ვარ ანა"
console.log(student2.introduce()); // ორივე instance-ი "იზიარებს" ერთსა და იმავე მეთოდს

მემკვიდრეობითობა (inheritance) საშუალებას გვაძლევს, ერთმა „ტიპმა" მემკვიდრეობით მიიღოს მეორის თვისებები და მეთოდები — ეს პრაქტიკაში დღეს თითქმის ყოველთვის class სინტაქსით (იხ. ქვემოთ) ხორციელდება.

3. Class-ები — თანამედროვე სინტაქსი

ES6-ის class არის „სინტაქსური შაქარი" ზემოთ ნაჩვენები constructor-function/prototype სისტემის თავზე — იგივე მექანიზმია, უფრო წასაკითხავი ფორმით: 

class Student {
  constructor(name, age) {
    this.name = name;
    this._age = age; // კონვენცია: "_" პრეფიქსი "შიდა" თვისებისთვის (getter/setter-თან)
  }

  introduce() {
    return `გამარჯობა, მე ვარ ${this.name}`;
  }
}

const student1 = new Student("ანა", 22);
console.log(student1.introduce());

მემკვიდრეობითობა Class-ებში: extends

class GraduateStudent extends Student {
  constructor(name, age, thesisTopic) {
    super(name, age); // მშობელი კლასის constructor-ის გამოძახება
    this.thesisTopic = thesisTopic;
  }

  introduce() {
    return `${super.introduce()}, ჩემი თემაა "${this.thesisTopic}"`;
  }
}

const grad = new GraduateStudent("მარი", 27, "Front-end განათლება");
console.log(grad.introduce());

4. Setters & Getters

Getter/setter გვაძლევს კონტროლს იმაზე, თუ როგორ იკითხება/იწერება თვისება — შესაძლებელია დამატებითი ლოგიკის (მაგ. ვალიდაციის) ჩამატება: 

class Student {
  constructor(name, age) {
    this.name = name;
    this._age = age;
  }

  get age() {
    return this._age;
  }

  set age(value) {
    if (value < 0) {
      throw new Error("ასაკი ვერ იქნება უარყოფითი");
    }
    this._age = value;
  }
}

const s = new Student("ანა", 22);
console.log(s.age);   // 22 — getter-ის გამოძახება, ფრჩხილების გარეშე
s.age = 23;           // setter-ის გამოძახება
s.age = -5;           // ❌ Error — setter-ის ვალიდაცია მუშაობს

5. instanceof ოპერატორი

ამოწმებს, არის თუ არა ობიექტი კონკრეტული class-ის (ან constructor-ის) instance-ი:

console.log(grad instanceof GraduateStudent); // true
console.log(grad instanceof Student);         // true — მემკვიდრეობის გამო
console.log(grad instanceof Array);           // false

6. მონაცემების შენახვა ბრაუზერში

ბრაუზერს აქვს რამდენიმე ჩაშენებული მექანიზმი, რომ მონაცემი „გადარჩეს" გვერდის ხელახლა ჩატვირთვის შემდეგაც: 

// localStorage — რჩება უსასრულოდ, სანამ ხელით არ წაიშლება
localStorage.setItem("theme", "dark");
console.log(localStorage.getItem("theme")); // "dark"
localStorage.removeItem("theme");

// sessionStorage — იშლება, როცა ბრაუზერის ტაბი იხურება
sessionStorage.setItem("draft", JSON.stringify({ text: "..." }));

// ობიექტების შენახვისას — ყოველთვის JSON.stringify/parse საჭიროა
const user = { name: "ანა", loggedIn: true };
localStorage.setItem("user", JSON.stringify(user));
const restoredUser = JSON.parse(localStorage.getItem("user"));

Cookies უფრო ძველი მექანიზმია — ავტომატურად იგზავნება ყოველ HTTP მოთხოვნასთან ერთად სერვერზე (localStorage/sessionStorage არ იგზავნება), რაც მას გამოსადეგს ხდის session/auth ინფორმაციისთვის, თუმცა ტევადობა ბევრად ნაკლებია: 

document.cookie = "username=ana; max-age=3600"; // 1 საათის ვადით
console.log(document.cookie); // "username=ana" — ყველა cookie ერთ სტრიქონად

7. პრაქტიკა: ფილტრის აწყობა

ყველაფერი ერთად — მონაცემთა მასივის ფილტრაცია (მე-12 გაკვეთილიდან), DOM (მე-15) და events (მე-16): 

const products = [
  { name: "წიგნი", category: "განათლება", price: 15 },
  { name: "ყურსასმენი", category: "ტექნიკა", price: 80 },
  { name: "რვეული", category: "განათლება", price: 5 },
];

const filterButtons = document.querySelectorAll("[data-filter]");
const list = document.querySelector("#product-list");

function render(items) {
  list.innerHTML = items
    .map(p => `<li>${p.name} — ${p.price}₾</li>`)
    .join("");
}

filterButtons.forEach(btn => {
  btn.addEventListener("click", () => {
    const category = btn.dataset.filter;
    const filtered = category === "all"
      ? products
      : products.filter(p => p.category === category);
    render(filtered);
  });
});

render(products); // საწყისი მდგომარეობა — ყველა პროდუქტი
