1. React Form და ვალიდაცია

React-ში ფორმის ველები ჩვეულებრივ არის controlled — მათი მნიშვნელობა ინახება state-ში, არა თავად DOM-ში (განსხვავებით ჩვეულებრივი HTML ფორმებისგან): 

function ContactForm() {
  const [email, setEmail] = useState("");
  const [error, setError] = useState("");

  function handleSubmit(event) {
    event.preventDefault(); // ისევე, როგორც vanilla JS-ში (მე-16 გაკვეთილი)

    if (!email.includes("@")) {
      setError("გთხოვთ, მიუთითოთ ვალიდური ელფოსტა");
      return;
    }
    setError("");
    console.log("იგზავნება:", email);
  }

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
      />
      {error && <p className="error">{error}</p>}
      <button type="submit">გაგზავნა</button>
    </form>
  );
}

value={"{email}"} + onChange წყვილი — ეს არის „controlled input"-ის ფორმულა: React state ყოველთვის „წყაროა ჭეშმარიტებისა" (source of truth) input-ის მიმდინარე მნიშვნელობისთვის.

რამდენიმე ველი ერთი handler-ით

როცა ფორმას 5–10 ველი აქვს, თითოეულისთვის ცალკე useState და ცალკე handler მოუხერხებელია. გავრცელებული ხერხია ერთი state ობიექტი და ერთი onChange, რომელიც ველის name ატრიბუტს იყენებს: 

const [values, setValues] = useState({ fullName: "", email: "", city: "Tbilisi", subscribe: false });

function handleChange(e) {
  const { name, type, value, checked } = e.target;
  setValues((prev) => ({
    ...prev,                                   // spread — უცვლელი (immutable) განახლება, მე-14 გაკვეთილიდან
    [name]: type === "checkbox" ? checked : value,   // [name] — computed property
  }));
}

<input name="fullName" value={values.fullName} onChange={handleChange} />
<input name="email" type="email" value={values.email} onChange={handleChange} />
<select name="city" value={values.city} onChange={handleChange}>
  <option value="Tbilisi">თბილისი</option>
  <option value="Batumi">ბათუმი</option>
</select>
<label>

</label>
  • text, email, number, textarea, select — value + onChange (React-ში <textarea value=...> და <select value=...> იგივე ლოგიკით მუშაობს).
  • checkbox — checked + onChange (იკითხება e.target.checked).
  • radio — თითოეულს checked={"{value === \"x\"}"} და ერთი და იგივე name.
  • type="number" ინპუტიც სტრიქონს გვიბრუნებს (e.target.value) — საჭიროა Number(...).
  • ფორმის გაგზავნის ღილაკს ყოველთვის დაუწერე type="submit", სხვა ღილაკებს — type="button" (თორემ ნებისმიერი ღილაკი ფორმას გააგზავნის).

ვალიდაცია: ფუნქცია, ერთი შეცდომების ობიექტი

კარგი პრაქტიკაა ვალიდაცია ცალკე წმინდა ფუნქციად გაიტანო — ის მიიღებს მნიშვნელობებს და დააბრუნებს შეცდომების ობიექტს: 

function validate(values) {
  const errors = {};
  if (values.fullName.trim().length < 3) errors.fullName = "სახელი და გვარი სავალდებულოა (მინ. 3 სიმბოლო)";
  if (!/^\S+@\S+\.\S+$/.test(values.email)) errors.email = "ელფოსტის ფორმატი არასწორია";
  if (!/^\+?\d{9,15}$/.test(values.phone)) errors.phone = "ტელეფონი: მხოლოდ ციფრები (9–15), სურვილისამებრ + დასაწყისში";
  return errors;           // ცარიელი ობიექტი = ყველაფერი რიგზეა
}

function CheckoutForm() {
  const [values, setValues] = useState({ fullName: "", email: "", phone: "" });
  const [touched, setTouched] = useState({});      // რომელი ველია უკვე „შეხებული“
  const [submitted, setSubmitted] = useState(false);

  const errors = validate(values);                // გამოითვლება რენდერისას — ცალკე state არ სჭირდება!
  const isValid = Object.keys(errors).length === 0;

  const showError = (name) => (touched[name] || submitted) && errors[name];

  function handleSubmit(e) {
    e.preventDefault();
    setSubmitted(true);
    if (!isValid) return;
    // ...გაგზავნა
  }

  return (
    <form onSubmit={handleSubmit} noValidate>
      <label htmlFor="fullName">სახელი და გვარი</label>
      <input
        id="fullName"
        name="fullName"
        value={values.fullName}
        onChange={handleChange}
        onBlur={(e) => setTouched((t) => ({ ...t, [e.target.name]: true }))}
        aria-invalid={Boolean(showError("fullName"))}
        aria-describedby={showError("fullName") ? "fullName-error" : undefined}
      />
      {showError("fullName") && <p id="fullName-error" role="alert">{errors.fullName}</p>}
      {/* ... დანარჩენი ველები ანალოგიურად ... */}
      <button type="submit">გაგზავნა</button>
    </form>
  );
}

ვალიდაციის UX წესები

  • როდის ვაჩვენოთ შეცდომა? არა პირველივე სიმბოლოზე (მომხმარებელი ჯერ წერს, შეცდომას უკვე „უყვირი“), არამედ ველის დატოვების (onBlur, „touched“) ან გაგზავნის შემდეგ. გასწორების შემდეგ კი შეცდომა დაუყოვნებლივ უნდა გაქრეს.
  • შეცდომის ტექსტი კონკრეტული და გამოსასწორებელია: არა „არასწორი მნიშვნელობა“, არამედ „ტელეფონი უნდა შეიცავდეს 9–15 ციფრს“.
  • ხელმისაწვდომობა: ყოველ ველს ჰქონდეს <label htmlFor> (არა მხოლოდ placeholder), შეცდომა aria-describedby-ით დაუკავშირდეს ველს, aria-invalid და შეცდომის ტექსტზე role="alert".
  • გაგზავნის დროს ღილაკი გამორთე და აჩვენე მდგომარეობა („იგზავნება...“), რომ ორმაგი დაწკაპუნებით ორჯერ არ გაიგზავნოს.
  • ჯერ HTML-ის ჩაშენებული ატრიბუტები: required, type="email", minLength, pattern, autoComplete (მაგ. autoComplete="email", "new-password") — ბრაუზერისა და პაროლ-მენეჯერებისთვის დიდი დახმარებაა. noValidate-ს მხოლოდ მაშინ დაუმატებ ფორმას, როცა საკუთარი შეტყობინებების ჩვენება გინდა.
  • კლიენტის ვალიდაცია მხოლოდ მოხერხებულობაა, არა უსაფრთხოება. ბრაუზერში ყველაფერი შეიძლება გვერდი აუარო; სერვერმა ყველა მონაცემი თავისით უნდა გადაამოწმოს. სერვერის შეცდომები ფორმაში აჩვენე (მაგ. „ეს ელფოსტა უკვე გამოყენებულია“).

React 19: ფორმის Actions — useActionState და useFormStatus

React 19-მა ფორმებს ახალი, უფრო დეკლარაციული მიდგომა შესძინა: <form>-ის action prop-ში შეიძლება ფუნქცია გადავცეთ. React ამ ფუნქციას ფორმის გაგზავნისას გამოიძახებს, გადასცემს FormData-ს, თვითონ აკონტროლებს „მიმდინარეობს“ მდგომარეობას და ფორმის (არაკონტროლირებად) ველებს წარმატებული დასრულების შემდეგ ავტომატურად ასუფთავებს. 

import { useActionState } from "react";          // React 19: react-დან (ადრე useFormState იყო react-dom-ში)
import { useFormStatus } from "react-dom";

function SubmitButton({ children }) {
  const { pending } = useFormStatus();            // მუშაობს მხოლოდ <form>-ის შვილ კომპონენტში!
  return <button type="submit" disabled={pending}>{pending ? "იგზავნება..." : children}</button>;
}

function NewsletterForm() {
  const [state, formAction, isPending] = useActionState(
    async (previousState, formData) => {          // (წინა state, FormData) => ახალი state
      const email = String(formData.get("email") ?? "").trim();

      if (!/^\S+@\S+\.\S+$/.test(email)) {
        return { ok: false, message: "ელფოსტა არასწორია", email };
      }
      await new Promise((r) => setTimeout(r, 800));   // აქ იქნებოდა API მოთხოვნა
      return { ok: true, message: `გამოწერილია: ${email}`, email: "" };
    },
    { ok: false, message: "", email: "" }          // საწყისი state
  );

  return (
    <form action={formAction}>
      <input name="email" type="email" defaultValue={state.email} aria-label="ელფოსტა" />
      <SubmitButton>გამოწერა</SubmitButton>
      {state.message && <p role="status">{state.message}</p>}
    </form>
  );
}

Actions — ნიუანსები, რომლებიც ხშირად ავიწყდებათ

  • useFormStatus() უნდა გამოიძახო ფორმის შიგნით მდებარე კომპონენტში (ცალკე SubmitButton). თუ იმავე კომპონენტში გამოიძახებ, სადაც <form> იხატება, pending ყოველთვის false იქნება.
  • წარმატებული action-ის შემდეგ React არაკონტროლირებად ველებს (defaultValue) ავტომატურად ასუფთავებს, ისევე, როგორც ჩვეულებრივი HTML ფორმა. იმ ველის მნიშვნელობის შესანარჩუნებლად, სადაც შეცდომა მოხდა, state-იდან დააბრუნე და defaultValue={"{state.email}"}-ით ჩასვი (როგორც ზემოთ).
  • controlled ველებს (value + onChange) action ავტომატურად არ ასუფთავებს — შენ თვითონ უნდა გაასუფთავო state (ან გამოიყენე requestFormReset from react-dom).
  • useOptimistic — კიდევ ერთი React 19 Hook, რომელიც სერვერის პასუხამდე შედეგის „ოპტიმისტურ“ ჩვენებას იძლევა (მაგ. კომენტარი მაშინვე ჩანს). ეს მოწინავე თემაა, ამ კურსში აღარ შევალთ.

React Hook Form + Zod — რთული ფორმებისთვის

როცა ფორმაში ბევრი ველი, პირობითი ლოგიკა და განმეორებადი წესებია, ხშირად იყენებენ ბიბლიოთეკებს. ყველაზე გავრცელებული კომბინაცია 2026-ში: React Hook Form (ველების მართვა — ის ძირითადად არაკონტროლირებად ინპუტებს იყენებს, ამიტომ ნაკლები re-render-ია) + Zod (ვალიდაციის „სქემა“, რომელიც ერთხელ ეწერება და ბევრგან მეორდება) + @hookform/resolvers (მათი დამაკავშირებელი). მცირე და მარტივი ფორმებისთვის კი React 19-ის ჩაშენებული Actions ხშირად საკმარისია.

const schema = z.object({
  email: z.email("ელფოსტა არასწორია"),
  password: z.string().min(8, "პაროლი მინიმუმ 8 სიმბოლოა"),

function SignUpForm() {

    defaultValues: { email: "", password: "" },
    mode: "onTouched",                        // შეცდომა — ველის დატოვების შემდეგ
  });

  async function onSubmit(values) {           // გამოიძახება მხოლოდ ვალიდური მონაცემებით
    await api.post("/signup", values);
  }

  return (
    <form onSubmit={handleSubmit(onSubmit)} noValidate>
      <input type="email" autoComplete="email" {...register("email")} />
      {errors.email && <p role="alert">{errors.email.message}</p>}

      <input type="password" autoComplete="new-password" {...register("password")} />

    </form>
  );
}

ვერსიების შენიშვნა: Zod 4-ში (2025 წლის ივლისიდან) შეცდომის შეტყობინებებს ერთიანი error პარამეტრი განსაზღვრავს; Zod 3-ის required_error და invalid_type_error ახალ ვერსიაში უბრალოდ აღარ მუშაობს (ჩუმად იგნორირდება). Zod 4-ში გამოიყენება მოკლე z.email() (ძველი z.string().email() ჯერ მუშაობს, მაგრამ მოძველებულია). თუ ძველი ტუტორიალის კოდი ისე არ მუშაობს, როგორც უნდა, ყველაზე ხშირი მიზეზი ეს არის.

2. საბაზისო ავტორიზაციის ლოგიკა

რეალურ პროექტში ავტორიზაცია მოიცავს სერვერს, ტოკენებს და უსაფრთხოების ბევრ დეტალს — მაგრამ საბაზისო, front-end მხარეს, პატერნი ასე გამოიყურება (მე-20 გაკვეთილის storage-ცოდნაზე დაფუძნებული): 

function LoginForm({ onLogin }) {
  const [username, setUsername] = useState("");

  function handleSubmit(event) {
    event.preventDefault();
    // რეალურ აპლიკაციაში აქ იქნებოდა API მოთხოვნა (axios.post("/login", ...))
    sessionStorage.setItem("user", JSON.stringify({ username }));
    onLogin(username); // მშობელი კომპონენტისთვის ცნობა, რომ ავტორიზაცია მოხდა
  }

  return (
    <form onSubmit={handleSubmit}>
      <input value={username} onChange={(e) => setUsername(e.target.value)} />
      <button type="submit">შესვლა</button>
    </form>
  );
}

„დაცული" (protected) გვერდები React Router-ში ჩვეულებრივ სტატუსს ამოწმებენ და საჭიროების შემთხვევაში ავტომატურად გადამისამართებენ login გვერდზე — ეს ლოგიკა ჩვეულებრივ იტანება ერთ, ხელახლა გამოსაყენებელ კომპონენტში.

ზემოთ მოცემული მაგალითი მხოლოდ „სათამაშოა“

ის არ ამოწმებს პაროლს და მხოლოდ UI-ს გადართავს. ქვემოთ დეტალურად ვნახავთ, როგორ გამოიყურება ეს „სერვერთან ერთად“ და რატომ უნდა იყოს ფრთხილად ის, თუ სად ვინახავთ ავტორიზაციის მონაცემებს.

ძირითადი ცნებები

  • ავთენტიფიკაცია (authentication) — „ვინ ხარ?“ (შესვლა: მომხმარებლის სახელი + პაროლი).
  • ავტორიზაცია (authorization) — „რისი უფლება გაქვს?“ (მაგ. მხოლოდ ადმინს შეუძლია პროდუქტის წაშლა). ამ ორ სიტყვას ხშირად ერთმანეთში ურევენ; ამ გაკვეთილზე „ავტორიზაციას“ სასაუბრო მნიშვნელობით, ანუ „შესვლის სისტემას“ ვეძახით.
  • სესია (cookie-ზე დაფუძნებული) — სერვერი იმახსოვრებს, ვინ შემოვიდა და ბრაუზერს აძლევს cookie-ს; ან ტოკენი (JWT და სხვ.) — სერვერი ხელმოწერილ ტოკენს აბრუნებს და ბრაუზერი მას ყოველ მოთხოვნაში Authorization: Bearer ... header-ით აგზავნის.

ტიპური შესვლის ნაკადი

  1. მომხმარებელი ავსებს ფორმას (ვალიდაცია კლიენტზე).
  2. აპლიკაცია აგზავნის POST /auth/login მოთხოვნას {"{ username, password }"} სხეულით (HTTPS-ით!).
  3. სერვერი ამოწმებს მონაცემებს: წარმატებისას აბრუნებს ტოკენს (ან ადგენს cookie-ს) და მომხმარებლის პროფილს; წარმატების გარეშე — 400/401.
  4. აპლიკაცია „ინახავს“ ავტორიზებულ მდგომარეობას (ქვემოთ — სად) და გადაჰყავს დაცულ გვერდზე ან იქ, საიდანაც მოვიდა.
  5. შემდეგ მოთხოვნებს აპლიკაცია ტოკენს ახლავს (Axios interceptor — მე-23 გაკვეთილი).
  6. გამოსვლისას (logout) აპლიკაცია ასუფთავებს მდგომარეობას და ტოკენს.

სად შევინახოთ ტოკენი? (უსაფრთხოების მნიშვნელოვანი დეტალი)

localStorage და sessionStorage ხელმისაწვდომია ნებისმიერი JavaScript-ისთვის გვერდზე. თუ საიტზე XSS ხარვეზი (ან დაზიანებული მესამე მხარის სკრიპტი) არსებობს, თავდამსხმელს მათი წაკითხვა შეუძლია. ამიტომ OWASP-ისა და OAuth-ის ბრაუზერ-აპლიკაციების რეკომენდაციები გრძელვადიანი ტოკენების localStorage-ში შენახვას არ ურჩევს.

ადგილიXSS-ით ქურდობაშენიშვნა
localStorageშესაძლებელიაგადარჩება ბრაუზერის დახურვას; არ არის HttpOnly ანალოგი
sessionStorageშესაძლებელიაიშლება ტაბის დახურვისას, მაგრამ XSS-ისგან არ გვიცავს
HttpOnly cookieJS ვერ კითხულობსსაჭიროა CSRF დაცვა (SameSite, CSRF ტოკენი); ტოკენს სერვერი ადგენს
JS მეხსიერება (state / მოდულის ცვლადი)ძალიან გაძნელებულიაგვერდის განახლებისას იკარგება → საჭიროა refresh ტოკენი HttpOnly cookie-ში

დღეს გავრცელებული რეკომენდაცია SPA-სთვის: მოკლევადიანი access ტოკენი — მეხსიერებაში, refresh ტოკენი — HttpOnly + Secure + SameSite cookie-ში (ტოკენების როტაციით), ან სრულიად „ტოკენის გარეშე“ სერვერული სესია cookie-ით (მაგ. BFF — Backend for Frontend — არქიტექტურით). ეს სერვერის მხარესაც ცვლილებებს მოითხოვს, ამიტომ სასწავლო პროექტში კომპრომისს ვიყენებთ და ამას ღიად ვაღიარებთ:

  • ტოკენს ვინახავთ მხოლოდ მეხსიერებაში (მოდულის ცვლადში, Axios interceptor-ისთვის).
  • sessionStorage-ში ვინახავთ მხოლოდ არასენსიტიურ პროფილს (id, firstName...) — რომ გვერდის განახლებისას UI „არ გაგვიხსნას“. პაროლს და ტოკენს არასდროს ვინახავთ.
  • გვერდის განახლების შემდეგ ტოკენი იკარგება — რეალურ აპლიკაციაში აქ refresh cookie იმუშავებდა.

Front-end „დაცვა“ UX-ია, არა უსაფრთხოება

<ProtectedRoute> მხოლოდ მომხმარებელს არ აჩვენებს გვერდს; ნებისმიერს შეუძლია JavaScript-ის შეცვლა ან API-ს უშუალო გამოძახება. ნამდვილი დაცვა ყოველთვის სერვერზეა: ყოველი დაცული API მოთხოვნა სერვერმა უნდა გადაამოწმოს (ტოკენი, როლი, უფლებები). ზუსტად ამიტომ, front-end-ში არასდროს ჩაწერო „ადმინის პაროლი“ ან საიდუმლო გასაღები და არ ენდო isAdmin მნიშვნელობას, რომელიც ბრაუზერში იცვლება.

უსაფრთხო ავტორიზაციის ჩეკლისტი (front-end)

  • ყოველთვის HTTPS; პაროლს არასდროს ვინახავთ localStorage-ში, არც URL-ში არ ვაგზავნით და არც კონსოლში ვბეჭდავთ (console.log(user) ტოკენის ბეჭდვამდეც შეიძლება მიგვიყვანოს).
  • პაროლის ველი: type="password", autoComplete="current-password" (შესვლისას) და "new-password" (რეგისტრაციისას) — პაროლ-მენეჯერები გამოიყენებენ.
  • შეცდომის ტექსტი ზოგადი იყოს: „სახელი ან პაროლი არასწორია“ (არ დააზუსტო, რომელია არასწორი — ეს მომხმარებლების ჩამოთვლის საშუალებას იძლევა).
  • ჩაშენებული გავრცელებული საფრთხეებისგან დაცვა: React ტექსტს ავტომატურად ექრანირებს (XSS) — dangerouslySetInnerHTML-ს ნუ გამოიყენებ მომხმარებლის შიგთავსზე; პაკეტებს აკონტროლებ (lock ფაილი, npm ci, npm audit).
  • რეალურ პროექტში თვითონ ნუ ააშენებ ავტორიზაციის სისტემას (პაროლების შენახვა, აღდგენა, 2FA, სოციალური შესვლა): გამოიყენე გამართული სერვისები და ბიბლიოთეკები (მაგ. Auth0, Clerk, Firebase Auth, Supabase Auth, Keycloak) ან სერვერის ფრეიმვორქის ჩაშენებული გადაწყვეტა.
  • სერვერი უნდა იცავდეს brute-force შეტევისგან (rate limiting) — ეს ეხება backend-ს, მაგრამ შენც უნდა იცოდე (429 სტატუსი).

AuthContext და დაცული მარშრუტი (მოკლედ)

ავტორიზაციის მდგომარეობა (ვინ არის შესული) ბევრ კომპონენტს სჭირდება: ნავიგაციას, დაცულ გვერდებს, შეკვეთის ფორმას. ეს Context-ის კლასიკური გამოყენებაა (იხ. შემდეგი სექცია). დაცული მარშრუტი — layout მარშრუტი, რომელიც ან შვილს აჩვენებს (<Outlet />), ან login-ზე გადამისამართებს და „გახსოვს, საიდან მოვიდა“: 

// components/ProtectedRoute.jsx
import { Navigate, Outlet, useLocation } from "react-router";
import { useAuth } from "../context/AuthContext";

export default function ProtectedRoute() {
  const { user } = useAuth();
  const location = useLocation();

  if (!user) {
    // state-ში ვინახავთ საიდან მოვიდა, რომ შესვლის შემდეგ უკან დავაბრუნოთ
    return <Navigate to="/login" replace state={{ from: location.pathname }} />;
  }
  return <Outlet />;
}

// App.jsx-ში:
<Route element={<ProtectedRoute />}>
  <Route path="checkout" element={<CheckoutPage />} />
</Route>

3. Context API და მდგომარეობის გაზიარება

Cart-ის მდგომარეობა (რომელი პროდუქტებია კალათაში) საჭიროებს „გაზიარებას" ბევრ კომპონენტს შორის (ProductCard-ს „დამატება" ღილაკიდან, Navbar-ს რაოდენობის ჩვენებამდე, Cart გვერდამდე) — ამისთვის React გვთავაზობს Context API-ს (props-ის ყოველ დონეზე ხელით გადაცემის (prop drilling) თავიდან ასაცილებლად): 

// CartContext.jsx — გამარტივებული მაგალითი
import { createContext, useState, useContext } from "react";

const CartContext = createContext();

export function CartProvider({ children }) {
  const [items, setItems] = useState([]);

  function addToCart(product) {
    setItems(prev => [...prev, product]); // spread — უცვლელი (immutable) განახლება, მე-14 გაკვეთილიდან
  }

  return (
    <CartContext value={{ items, addToCart }}>
      {children}
    </CartContext>
  );
}

export function useCart() {
  return useContext(CartContext); // custom hook, კონტექსტთან წვდომისთვის
}
// ProductCard.jsx — ნებისმიერ სიღრმეზე, props-ის გადაცემის გარეშე
function ProductCard({ product }) {
  const { addToCart } = useCart();
  return (
    <div className="card">
      <h3>{product.name}</h3>
      <button onClick={() => addToCart(product)}>კალათაში დამატება</button>
    </div>
  );
}

Context-ის სამი ნაბიჯი

  1. შექმნა: const CartContext = createContext(null).
  2. მიწოდება: ხის ზედა ნაწილში <CartContext value={"{...}"}>-ით. React 19-ში კონტექსტის ობიექტი თავად არის provider; ძველ კოდში ხშირად გაქვს <CartContext.Provider value=...> — ის კვლავ მუშაობს (მაგრამ მომავალ ვერსიაში მოძველებულად გამოცხადდება), ამიტომ ახალ კოდში გამოიყენე მოკლე ფორმა.
  3. წაკითხვა: useContext(CartContext) (ან React 19-ის use(CartContext), რომელიც პირობითადაც შეიძლება გამოიძახო).

უკეთესი პატერნი: ნულოვანი ნაგულისხმევი მნიშვნელობა და „დაცვა“

createContext() ცარიელად undefined-ს აბრუნებს, თუ provider-ის გარეთ გამოიყენე — შეცდომა მოგვიანებით და გაუგებრად გამოჩნდება („Cannot destructure property of undefined“). უკეთესია null-ით შექმნა და custom hook-ში აშკარა შეცდომის ამოგდება: 

const CartContext = createContext(null);

export function useCart() {
  const ctx = useContext(CartContext);
  if (ctx === null) {
    throw new Error("useCart უნდა გამოიყენო <CartProvider>-ის შიგნით");
  }
  return ctx;
}

რთული state: Context + useReducer

კალათას რამდენიმე მოქმედება აქვს: დამატება (ან რაოდენობის გაზრდა, თუ პროდუქტი უკვე არის), რაოდენობის შეცვლა, წაშლა, გასუფთავება. როცა ერთ state-ზე ბევრი სახის განახლება გვაქვს, useReducer ლოგიკას ერთ ფუნქციაში (reducer) აგროვებს და აადვილებს ტესტირებას: 

function cartReducer(items, action) {
  switch (action.type) {
    case "added": {
      const existing = items.find((i) => i.id === action.item.id);
      if (existing) {
        return items.map((i) =>
          i.id === action.item.id
            ? { ...i, quantity: Math.min(i.quantity + action.quantity, i.stock) }
            : i
        );
      }
      return [...items, { ...action.item, quantity: action.quantity }];
    }
    case "removed":
      return items.filter((i) => i.id !== action.id);
    case "cleared":
      return [];
    default:
      throw new Error("უცნობი action: " + action.type);
  }
}

const [items, dispatch] = useReducer(cartReducer, []);
dispatch({ type: "added", item: product, quantity: 1 });

reducer წმინდა ფუნქციაა (ისევე, როგორც კომპონენტი): იღებს მიმდინარე state-ს და action-ს, აბრუნებს ახალ state-ს, არაფერს „ცვლის“ და გარე სისტემასთან (API, localStorage) არ ურთიერთობს.

Context-ის შეზღუდვები

  • როცა context-ის value იცვლება, ყველა მომხმარებელი კომპონენტი ხელახლა იხატება. თუ value-ში ობიექტს რენდერში ახლად ქმნი (value={"{{ items, addToCart }}"}) და provider-ის მშობელი ხშირად რენდერდება, ეს ზედმეტ განახლებებს იწვევს — გამოიყენე useMemo (ან მხოლოდ სტაბილური ფუნქციები).
  • Context ხშირად ცვალებადი მონაცემისთვის (მაუსის პოზიცია, ანიმაცია) ცუდი არჩევანია.
  • დიდ აპლიკაციებში ხშირად იყენებენ სპეციალიზებულ ბიბლიოთეკებს (Zustand, Redux Toolkit, Jotai), ხოლო სერვერის მონაცემებისთვის — TanStack Query. პატარა-საშუალო პროექტისთვის Context + useReducer საკმარისია.
  • ცალკე სფეროებისთვის ცალკე context-ები გამოიყენე (AuthContext, CartContext, ThemeContext) და არა ერთი „გიგანტური“.

import { useState, useEffect, useCallback } from "react";
import axios from "axios";
import { api, getErrorMessage } from "../api/client";

export function useFetch(url) {
  const [state, setState] = useState({ data: null, loading: Boolean(url), error: null });
  const [reloadCount, setReloadCount] = useState(0);

  const reload = useCallback(() => setReloadCount((c) => c + 1), []);

  useEffect(() => {
    if (!url) return;                                   // URL ჯერ არ არის
    const controller = new AbortController();

    setState((prev) => ({ ...prev, loading: true, error: null }));

    api
      .get(url, { signal: controller.signal })
      .then((res) => setState({ data: res.data, loading: false, error: null }))
      .catch((err) => {
        if (axios.isCancel(err)) return;                // გაუქმება შეცდომა არ არის
        setState({ data: null, loading: false, error: getErrorMessage(err) });
      });

    return () => controller.abort();                    // cleanup: ძველი მოთხოვნის გაუქმება
  }, [url, reloadCount]);

  return { ...state, reload };
}

// src/context/CartContext.jsx
import { createContext, useContext, useEffect, useMemo, useReducer } from "react";
import { readJSON, writeJSON } from "../utils/storage";
import { getFinalPrice } from "../utils/money";

const CartContext = createContext(null);
const CART_KEY = "shophub:cart";

function cartReducer(items, action) {
  switch (action.type) {
    case "added": {
      const { product, quantity } = action;
      if (product.stock < 1) return items;                    // მარაგში არ არის
      const existing = items.find((i) => i.id === product.id);

      if (existing) {
        return items.map((i) =>
          i.id === product.id
            ? { ...i, quantity: Math.min(i.quantity + quantity, i.stock) }
            : i
        );
      }
      return [
        ...items,
        {
          id: product.id,
          title: product.title,
          thumbnail: product.thumbnail,
          price: getFinalPrice(product.price, product.discountPercentage),
          stock: product.stock,
          quantity: Math.min(quantity, product.stock),
        },
      ];
    }
    case "quantity_set":
      return items.map((i) =>
        i.id === action.id
          ? { ...i, quantity: Math.max(1, Math.min(action.quantity, i.stock)) }
          : i
      );
    case "removed":
      return items.filter((i) => i.id !== action.id);
    case "cleared":
      return [];
    default:
      throw new Error("უცნობი action: " + action.type);
  }
}

export function CartProvider({ children }) {
  // მესამე არგუმენტი — ლენივი ინიციალიზაცია: localStorage-ის წაკითხვა მხოლოდ ერთხელ
  const [items, dispatch] = useReducer(cartReducer, null, () => readJSON(localStorage, CART_KEY, []));

  useEffect(() => {
    writeJSON(localStorage, CART_KEY, items);      // გვერდითი ეფექტი — ეფექტში, არა reducer-ში
  }, [items]);

  const value = useMemo(
    () => ({
      items,
      totalItems: items.reduce((sum, i) => sum + i.quantity, 0),
      totalPrice: items.reduce((sum, i) => sum + i.price * i.quantity, 0),
      addItem: (product, quantity = 1) => dispatch({ type: "added", product, quantity }),
      setQuantity: (id, quantity) => dispatch({ type: "quantity_set", id, quantity }),
      removeItem: (id) => dispatch({ type: "removed", id }),
      clearCart: () => dispatch({ type: "cleared" }),
    }),
    [items]
  );

  return <CartContext value={value}>{children}</CartContext>;
}

export function useCart() {
  const ctx = useContext(CartContext);
  if (ctx === null) throw new Error("useCart უნდა გამოიყენო <CartProvider>-ის შიგნით");
  return ctx;
}

const AuthContext = createContext(null);

export function AuthProvider({ children }) {

    try {
      const { data } = await api.post("/auth/login", { username, password, expiresInMins: 30 });

      setAccessToken(data.accessToken);              // მოდულის ცვლადში (interceptor-ისთვის)

        username: data.username,
        firstName: data.firstName,
        lastName: data.lastName,
        email: data.email,
      };
      writeJSON(sessionStorage, USER_KEY, profile);
      setUser(profile);
      return profile;

      // 400 — არასწორი მონაცემები: ზოგადი შეტყობინება (არ ვაზუსტებთ რომელია არასწორი)

      }
      throw new Error(getErrorMessage(err));
    }
  }, []);

  const logout = useCallback(() => {
    setAccessToken(null);
    sessionStorage.removeItem(USER_KEY);
    setUser(null);
  }, []);

  const value = useMemo(
    () => ({ user, isAuthenticated: user !== null, login, logout }),
    [user, login, logout]
  );

  return <AuthContext value={value}>{children}</AuthContext>;
}

function SubmitButton() {
  const { pending } = useFormStatus();                     // მუშაობს, რადგან <form>-ის შიგნითაა
  return (

export default function LoginPage() {
  const { login } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();
  const from = location.state?.from ?? "/";

  const [state, formAction] = useActionState(
    async (_previousState, formData) => {
      const username = String(formData.get("username") ?? "").trim();
      const password = String(formData.get("password") ?? "");

      // 1) ვალიდაცია

        await login(username, password);
        navigate(from, { replace: true });                 // წარმატების შემდეგ დაბრუნება
        return { errors: {}, message: "", username };
      } catch (err) {

      }

          defaultValue={state.username}                  // შეცდომის შემდეგაც შენარჩუნდება

ველის მნიშვნელობა ინახება React state-ში (value + onChange), ანუ state არის ჭეშმარიტების ერთადერთი წყარო, და არა DOM.

2. როდის უნდა ვაჩვენოთ ვალიდაციის შეცდომა და რატომ?

ველის დატოვების (blur) ან გაგზავნის შემდეგ — არა პირველივე სიმბოლოზე. გასწორებისას კი შეცდომა დაუყოვნებლივ უნდა გაქრეს. ასე მომხმარებელი არ იგრძნობს „შეტევას“.

3. რას აკეთებს useActionState და რატომ უნდა იყოს useFormStatus ფორმის შიგნით არსებულ კომპონენტში?

useActionState ფორმის action-ს აკავშირებს state-თან და აბრუნებს [state, action, isPending]-ს. useFormStatus მშობელი ფორმის სტატუსს კითხულობს, ამიტომ ის მხოლოდ ფორმის შვილ კომპონენტში მუშაობს; იმავე კომპონენტში, სადაც <form> იხატება, ყოველთვის pending: false იქნება.

4. რატომ არ არის კლიენტის ვალიდაცია უსაფრთხოების საშუალება?

ბრაუზერში ყველაფერი შეიძლება შეიცვალოს ან გვერდი აუარო (API-ს უშუალო გამოძახებით). ვალიდაცია კლიენტზე მხოლოდ მოხერხებულობაა; სერვერმა ყველა მონაცემი თავისით უნდა გადაამოწმოს.

5. რატომ არ არის რეკომენდებული გრძელვადიანი ტოკენის localStorage-ში შენახვა და რა არის ალტერნატივა?

localStorage ხელმისაწვდომია გვერდზე გაშვებული ნებისმიერი JavaScript-ისთვის, ამიტომ XSS-ის შემთხვევაში ტოკენის მოპარვა შესაძლებელია. ალტერნატივა: მოკლევადიანი access ტოკენი მეხსიერებაში და refresh ტოკენი HttpOnly+Secure+SameSite cookie-ში (ან სერვერული სესია cookie-ით).

6. რას იცავს ProtectedRoute და რას არა?

ის მხოლოდ UI-ს მართავს (არ აჩვენებს გვერდს და გადამისამართებს). ნამდვილ უსაფრთხოებას სერვერი უზრუნველყოფს — ყოველი დაცული მოთხოვნა სერვერზე უნდა გადამოწმდეს.

7. როდის გამოვიყენოთ Context და რა არის მისი მთავარი ნაკლი?

როცა ერთი მონაცემი ბევრ დაშორებულ კომპონენტს სჭირდება და prop drilling მოუხერხებელია (მომხმარებელი, კალათა, თემა). ნაკლი: value-ს ცვლილებისას ყველა მომხმარებელი კომპონენტი ხელახლა იხატება; ხშირად ცვალებად მონაცემებზე არ გამოდგება.

8. რატომ ვიყენებთ კალათისთვის useReducer-ს?

ერთ state-ზე რამდენიმე სახის ცვლილებაა (დამატება, რაოდენობა, წაშლა, გასუფთავება); reducer მათ ერთ წმინდა ფუნქციაში აერთიანებს, რაც ლოგიკას გასაგებს და ტესტირებადს ხდის.

9. რატომ ვინახავთ ძებნას/გვერდს URL-ში და არა state-ში?

ბმული გასაზიარებელი ხდება, „უკან“ ღილაკი სწორად მუშაობს და გვერდის განახლებისას მდგომარეობა არ იკარგება.

10. რა სიფრთხილე გვმართებს საჯარო სატესტო API-ებთან (DummyJSON და მსგავსი)?

მათ არასოდეს ვუგზავნით რეალურ პერსონალურ ან გადახდის მონაცემებს; ისინი მონაცემებს რეალურად არ ინახავენ და შეიძლება შეიცვალონ ან მიუწვდომელი გახდნენ.
